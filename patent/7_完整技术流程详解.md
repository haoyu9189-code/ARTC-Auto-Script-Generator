```python
# script_generator.py

class ScriptGenerator:
    """Abaqus仿真脚本自动生成器"""

    def generate_all_scripts(self, params):
        """
        批量生成所有参数组合的仿真脚本

        输入: 参数空间定义
        输出: 预处理脚本、后处理脚本、批处理脚本
        """
        total_count = 0

        for cell_type in params["cell_types"]:
            for size in params["cell_sizes"]:
                for radius in params["strut_radii"]:
                    for slider in params["sliders"]:
                        for analysis in params["analysis_types"]:

                            # 生成单组参数的脚本
                            self._generate_single_script(
                                cell_type, size, radius, slider, analysis
                            )
                            total_count += 1

        return total_count

    def _generate_single_script(self, cell_type, size, radius, slider, analysis):
        """生成单组参数的仿真脚本"""

        # 1. 创建目录结构
        script_dir = self._create_directory(cell_type, size, radius, slider, analysis)

        # 2. 选择模板
        if analysis.startswith("Sta"):
            template = self._load_template("Static_model.py")
        else:
            template = self._load_template("Dynamic_model.py")

        # 3. 参数替换
        content = self._replace_parameters(template, {
            "cell_type": cell_type,
            "cell_size": size,
            "strut_radius": radius,
            "slider": slider,
            "analysis_type": analysis
        })

        # 4. 动态接触面检测代码注入
        content = self._inject_contact_detection(content, size)

        # 5. 网格密度自适应
        mesh_size = self._calculate_mesh_size(radius)
        content = content.replace("{{MESH_SIZE}}", str(mesh_size))

        # 6. 生成预处理脚本
        preprocess_script = self._generate_preprocess(content)
        self._save_script(script_dir, f"{self._get_job_name()}_preprocess.py", preprocess_script)

        # 7. 生成后处理脚本
        postprocess_script = self._generate_postprocess(analysis)
        self._save_script(script_dir, f"{self._get_job_name()}_postprocess.py", postprocess_script)
```

### 3.3 动态接触面检测代码

```python
# 注入到预处理脚本中的接触面检测代码

CONTACT_DETECTION_CODE = '''
def detect_contact_faces(part, cell_size, tolerance=0.01):
    """
    动态接触面检测

    原理:
    1. 根据cell_size计算目标Z坐标
    2. 遍历所有面，执行三重检测
    3. 兜底策略确保始终返回有效结果
    """
    target_z_top = cell_size / 2.0
    target_z_bottom = -cell_size / 2.0

    top_faces = []
    bottom_faces = []

    for face in part.faces:
        # 获取面的几何信息
        vertices = face.getVertices()
        centroid = face.pointOn[0]
        normal = face.getNormalFromPointArray([centroid])

        # ===== 顶面检测 =====
        # 检测1: 顶点Z坐标
        vertex_match = any(abs(v.pointOn[0][2] - target_z_top) < tolerance
                          for v in vertices)
        # 检测2: 面中心Z坐标
        center_match = abs(centroid[2] - target_z_top) < tolerance
        # 检测3: 法向量朝上
        normal_valid = normal[0][2] >= -0.1

        if (vertex_match or center_match) and normal_valid:
            top_faces.append(face)

        # ===== 底面检测 =====
        vertex_match_bot = any(abs(v.pointOn[0][2] - target_z_bottom) < tolerance
                               for v in vertices)
        center_match_bot = abs(centroid[2] - target_z_bottom) < tolerance
        normal_valid_bot = normal[0][2] <= 0.1

        if (vertex_match_bot or center_match_bot) and normal_valid_bot:
            bottom_faces.append(face)

    # ===== 兜底策略 =====
    if not top_faces:
        # 选择Z坐标最大的面
        all_faces_z = [(f, f.pointOn[0][2]) for f in part.faces]
        all_faces_z.sort(key=lambda x: x[1], reverse=True)
        top_faces = [all_faces_z[0][0]]

    if not bottom_faces:
        all_faces_z = [(f, f.pointOn[0][2]) for f in part.faces]
        all_faces_z.sort(key=lambda x: x[1])
        bottom_faces = [all_faces_z[0][0]]

    return top_faces, bottom_faces

# 调用检测
top_faces, bottom_faces = detect_contact_faces(lattice_part, cell_size)
'''
```

# 生成的后处理脚本结构

"""
{JOB_NAME}_postprocess.py

功能: 从.odb文件提取力-位移数据
"""

from odbAccess import *
from abaqusConstants import *

job_name = "BCC_5_0p35_4_StaCompre"
odb_path = job_name + ".odb"

# ===== 1. 打开ODB文件 =====
odb = openOdb(path=odb_path, readOnly=True)

# ===== 2. 智能变量查找 =====
def find_output_region(odb, priority_list=['RIGIDPLATE-2', 'RIGIDPLATE', 'MERGEDSTRUCTURE']):
    """
    按优先级查找输出区域
    """
    step = odb.steps['Step-1']

    for priority in priority_list:
        for region_key in step.historyRegions.keys():
            if priority in region_key.upper():
                outputs = step.historyRegions[region_key].historyOutputs

                # 验证包含所需变量
                has_rf = any('RF' in k for k in outputs.keys())
                has_u = any('U' in k for k in outputs.keys())

                if has_rf and has_u:
                    return region_key, outputs

    return None, None

region_key, outputs = find_output_region(odb)

# ===== 3. 提取力-位移数据 =====
# 确定变量名
if 'StaCompre' in job_name or 'DynaCompre' in job_name:
    disp_var = 'U2'   # Y方向位移
    force_var = 'RF2'  # Y方向反力
else:  # Shear模式
    disp_var = 'U1'   # X方向位移
    force_var = 'RF1'  # X方向反力

# 查找精确变量名
for key in outputs.keys():
    if disp_var in key:
        disp_key = key
    if force_var in key:
        force_key = key

# 提取数据
disp_data = outputs[disp_key].data
force_data = outputs[force_key].data

# ===== 4. 组合并保存数据 =====
with open('feature_data.txt', 'w') as f:
    f.write(f"job_name: {job_name}\n")
    f.write(f"density: {open('density_temp.txt').read().strip()}\n")
    f.write(f"disp_var: {disp_key}\n")
    f.write(f"force_var: {force_key}\n")
    f.write("--- xy_combined data ---\n")

    for i in range(len(disp_data)):
        time, disp = disp_data[i]
        _, force = force_data[i]
        f.write(f"{abs(disp):.6f}\t{abs(force):.6f}\n")

# ===== 5. 关闭ODB =====
odb.close()

print("Postprocessing completed. Data saved to feature_data.txt")

#!/bin/bash
#PBS -N SmartAM_BCC_Batch

#PBS -l select=1:ncpus=8:mem=16gb
#PBS -l walltime=168:00:00

cd $PBS_O_WORKDIR

# ===== 脚本目录列表 =====
SCRIPT_DIRS=(
    "/home/user/generate_script/BCC/5/0p3/0/StaCompre"
    "/home/user/generate_script/BCC/5/0p3/1/StaCompre"
    "/home/user/generate_script/BCC/5/0p3/2/StaCompre"
    # ... 所有目录
)

TOTAL=${#SCRIPT_DIRS[@]}
COMPLETED=0
SKIPPED=0

echo "========================================"
echo "Smart AM Batch Execution"
echo "Total tasks: $TOTAL"
echo "Start time: $(date)"
echo "========================================"

# ===== 主循环 =====
for i in "${!SCRIPT_DIRS[@]}"; do
    script_dir="${SCRIPT_DIRS[$i]}"
    task_num=$((i + 1))

    echo ""
    echo "[$task_num/$TOTAL] Processing: $script_dir"

    cd "$script_dir"

    # ----- 智能断点续算判定 -----
    feature_file="feature_data.txt"

    if [ -f "$feature_file" ]; then
        file_size=$(wc -c < "$feature_file")

        if [ "$file_size" -ge 2000 ]; then
            # 提取最大位移
            max_disp=$(python3 -c "
import sys
try:
    with open('$feature_file') as f:
        lines = f.readlines()[5:]  # 跳过头部
        disps = [float(line.split()[0]) for line in lines if line.strip()]
        print(max(disps) if disps else 0)
except:
    print(0)
")

            if (( $(echo "$max_disp >= 0.8" | bc -l) )); then
                echo "  [SKIP] Already completed (disp=$max_disp >= 0.8)"
                ((SKIPPED++))
                continue
            fi
        fi
    fi

    # ----- 三阶段执行 -----
    job_name=$(basename "$script_dir" | sed 's/_StaCompre//')

    # Phase 1: 预处理
    echo "  [Phase 1] Preprocessing..."
    preprocess_script=$(ls *_preprocess.py 2>/dev/null | head -1)
    if [ -n "$preprocess_script" ]; then
        abaqus cae noGUI="$preprocess_script" > preprocess.log 2>&1
    fi

    # Phase 2: 求解
    echo "  [Phase 2] Solving..."
    inp_file=$(ls *.inp 2>/dev/null | head -1)
    if [ -n "$inp_file" ]; then
        job_base="${inp_file%.inp}"
        abaqus job="$job_base" input="$inp_file" cpus=8 > solver.log 2>&1

        # 等待求解完成
        while [ -f "${job_base}.lck" ]; do
            sleep 30
        done
    fi

    # Phase 3: 后处理
    echo "  [Phase 3] Postprocessing..."
    postprocess_script=$(ls *_postprocess.py 2>/dev/null | head -1)
    if [ -n "$postprocess_script" ]; then
        abaqus cae noGUI="$postprocess_script" > postprocess.log 2>&1
    fi

    # 验证结果
    if [ -f "feature_data.txt" ] && [ $(wc -c < "feature_data.txt") -ge 2000 ]; then
        echo "  [SUCCESS] Data extracted"
        ((COMPLETED++))
    else
        echo "  [FAILED] Data extraction failed"
    fi

done

# ===== 汇总报告 =====
echo ""
echo "========================================"
echo "Execution Summary"
echo "========================================"
echo "Total tasks:     $TOTAL"
echo "Completed:       $COMPLETED"
echo "Skipped:         $SKIPPED"
echo "Failed:          $((TOTAL - COMPLETED - SKIPPED))"
echo "End time:        $(date)"
echo "========================================"
```

# GeJsonl.py

class DataStandardizer:
    """数据标准化处理器"""

    def process_all_data(self, root_dir):
        """
        递归处理所有feature_data.txt文件

        输出: feature_data.json
        """
        all_data = {}

        for txt_file in self._find_all_feature_files(root_dir):
            sample_name, curve_data = self._process_single_file(txt_file)

            if sample_name not in all_data:
                all_data[sample_name] = {}

            all_data[sample_name].update(curve_data)

        # 保存JSON
        with open('feature_data.json', 'w') as f:
            json.dump(all_data, f, indent=2)

    def _process_single_file(self, txt_file):
        """处理单个feature_data.txt"""

        # 1. 解析元数据
        with open(txt_file, 'r') as f:
            lines = f.readlines()

        job_name = lines[0].split(': ')[1].strip()
        density = float(lines[1].split(': ')[1].strip())

        # 提取样本名和分析类型
        # BCC_5_0p35_4_StaCompre -> sample=BCC_5_0p35_4, analysis=StaCompre
        parts = job_name.rsplit('_', 1)
        sample_name = parts[0]
        analysis_type = parts[1]

        # 2. 读取力-位移数据
        data_start = next(i for i, line in enumerate(lines) if '---' in line) + 1
        raw_data = []
        for line in lines[data_start:]:
            if line.strip():
                disp, force = map(float, line.split())
                raw_data.append((disp, force))

        displacement = np.array([d[0] for d in raw_data])
        force = np.array([d[1] for d in raw_data])

        # 3. 收敛检测 - 删除尾部冗余数据
        convergence_idx = self._detect_convergence(displacement)
        displacement = displacement[:convergence_idx]
        force = force[:convergence_idx]

        # 4. 均匀插值
        if analysis_type.startswith('Sta'):
            target_points = 50   # 静态分析: 50点
        else:
            target_points = 500  # 动态分析: 500点

        x_uniform, y_uniform = self._uniform_interpolate(
            displacement, force, target_points
        )

        # 5. 组织输出
        curve_key = f"{analysis_type}_curve"
        return sample_name, {
            curve_key: {
                "displacement": x_uniform.tolist(),
                "force": y_uniform.tolist()
            },
            "density": density
        }

    def _detect_convergence(self, displacement, threshold=0.001, window=5):
        """
        检测位移收敛点

        原理: 当连续window个点的位移增量都小于threshold时，判定为收敛
        """
        for i in range(len(displacement) - window):
            increments = np.diff(displacement[i:i+window+1])
            if np.all(np.abs(increments) < threshold):
                return i + 1
        return len(displacement)

    def _uniform_interpolate(self, x, y, n_points):
        """均匀插值到n_points个点"""
        x_uniform = np.linspace(x.min(), x.max(), n_points)
        f_interp = interp1d(x, y, kind='linear', fill_value='extrapolate')
        y_uniform = f_interp(x_uniform)
        return x_uniform, y_uniform
```

### 6.1 力学特征提取

```python
# work/feature_utils.py

class FeatureExtractor:
    """力学特征提取器"""

    def extract_all_features(self, curve_data):
        """
        从力-位移曲线提取所有力学特征

        输入: {"displacement": [...], "force": [...]}
        输出: 特征字典
        """
        disp = np.array(curve_data["displacement"])
        force = np.array(curve_data["force"])

        # 转换为应力-应变
        strain = disp / self.cell_size  # 工程应变
        stress = force / self.cross_area  # 工程应力 (MPa)

        features = {}

        # ===== 基础特征 =====
        features["peak_stress"] = np.max(stress)
        features["peak_strain"] = strain[np.argmax(stress)]

        # ===== 刚度 (弹性模量) =====
        features["stiffness"] = self._calculate_stiffness(strain, stress)

        # ===== 屈服点 =====
        yield_strain, yield_stress = self._find_yield_point(strain, stress)
        features["yield_strain"] = yield_strain
        features["yield_stress"] = yield_stress

        # ===== 压实点 =====
        densification_strain = self._detect_densification(strain, stress, yield_stress)
        features["densification_strain"] = densification_strain

        # ===== 能量吸收 =====
        features["EA"] = self._calculate_energy_absorption(strain, stress, densification_strain)
        features["SEA"] = features["EA"] / self.mass  # 比吸能 (J/g)

        # ===== 平台应力 =====
        features["plateau_stress"] = self._calculate_plateau_stress(
            strain, stress, yield_strain, densification_strain
        )

        return features

    def _calculate_stiffness(self, strain, stress, fit_range=0.02):
        """
        计算刚度（弹性模量）

        方法: 在应变0-2%范围内线性拟合
        """
        mask = strain <= fit_range
        if np.sum(mask) < 3:
            mask = strain <= strain[5]  # 至少取前5个点

        # 线性回归
        coeffs = np.polyfit(strain[mask], stress[mask], 1)
        return coeffs[0]  # 斜率即为刚度

    def _find_yield_point(self, strain, stress, offset=0.002):
        """
        计算屈服点（0.2%偏移法）

        方法:
        1. 计算弹性模量E
        2. 画偏移线: σ = E × (ε - 0.002)
        3. 找偏移线与曲线的交点
        """
        E = self._calculate_stiffness(strain, stress)

        # 偏移线
        offset_line = E * (strain - offset)

        # 找交点（曲线首次低于偏移线的位置）
        diff = stress - offset_line

        # 在弹性区之后寻找
        search_start = np.argmax(strain > 0.01)

        for i in range(search_start, len(diff)):
            if diff[i] < 0:
                # 线性插值找精确交点
                t = diff[i-1] / (diff[i-1] - diff[i])
                yield_strain = strain[i-1] + t * (strain[i] - strain[i-1])
                yield_stress = stress[i-1] + t * (stress[i] - stress[i-1])
                return yield_strain, yield_stress

        # 未找到，返回峰值点
        peak_idx = np.argmax(stress)
        return strain[peak_idx], stress[peak_idx]

    def _detect_densification(self, strain, stress, yield_stress=None):
        """
        压实点检测

        算法:
        1. 在strain > 0.35范围内搜索
        2. 计算二阶导数d²σ/dε²
        3. 找首次超过自适应阈值的位置
        """
        # 均匀重采样
        strain_uniform = np.linspace(strain.min(), strain.max(), 100)
        stress_interp = interp1d(strain, stress, kind='cubic')
        stress_uniform = stress_interp(strain_uniform)

        # 平滑滤波
        stress_smooth = uniform_filter1d(stress_uniform, size=5)

        # 计算导数
        d_strain = strain_uniform[1] - strain_uniform[0]
        d1 = np.gradient(stress_smooth, d_strain)
        d2 = np.gradient(d1, d_strain)

        # 搜索范围: strain > 0.35
        search_mask = strain_uniform > 0.35

        # 限制应力范围 (避免密实区干扰)
        if yield_stress:
            search_mask &= stress_smooth < 2 * yield_stress

        if not np.any(search_mask):
            return strain_uniform[-1] * 0.7  # 兜底

        # 自适应阈值
        d2_search = d2[search_mask]
        threshold = np.max(d2_search) * 0.3

        # 找首次超过阈值的位置
        above_threshold = np.where(d2_search >= threshold)[0]

        if len(above_threshold) > 0:
            search_indices = np.where(search_mask)[0]
            densification_idx = search_indices[above_threshold[0]]
            return strain_uniform[densification_idx]

        # 兜底: 1.5倍屈服应力
        if yield_stress:
            idx = np.argmin(np.abs(stress_smooth - 1.5 * yield_stress))
            return strain_uniform[idx]

        return strain_uniform[-1] * 0.7

    def _calculate_energy_absorption(self, strain, stress, end_strain):
        """
        计算能量吸收

        EA = ∫σdε (从0到压实点)
        """
        mask = strain <= end_strain
        return np.trapz(stress[mask], strain[mask])

    def _calculate_plateau_stress(self, strain, stress, start_strain, end_strain):
        """
        计算平台应力

        平台应力 = EA / (ε_d - ε_y)
        """
        mask = (strain >= start_strain) & (strain <= end_strain)
        if np.sum(mask) < 2:
            return np.mean(stress)

        ea = np.trapz(stress[mask], strain[mask])
        plateau_length = end_strain - start_strain

        return ea / plateau_length if plateau_length > 0 else np.mean(stress[mask])
```

def extract_all_features_to_csv():
    """
    从feature_data.json提取所有特征，保存为CSV

    输出: extracted_features.csv (820+行 × 15+列)
    """
    with open('feature_data.json', 'r') as f:

        # 解析样本参数

        cell_type = parts[0]
        cell_size = int(parts[1])
        radius = float(parts[2].replace('p', '.'))

            "sample_name": sample_name,

    df = pd.DataFrame(rows)

class FeatureSearchWidget:
    """特征搜索与分析组件"""

    def __init__(self):
        self.feature_df = pd.read_csv('extracted_features.csv')

    def single_feature_search(self, feature_name, target_value, top_k=3):
        """
        单特征检索

        输入: 特征名, 目标值
        输出: Top-K最接近的材料
        """
        df = self.feature_df.copy()

        # 计算特征范围
        f_min = df[feature_name].min()
        f_max = df[feature_name].max()
        f_range = f_max - f_min

        # 归一化距离
        df['distance'] = abs(df[feature_name] - target_value) / f_range

        # 排序返回Top-K
        result = df.nsmallest(top_k, 'distance')

        return result[['sample_name', feature_name, 'distance']]

    def dual_feature_search(self, feat1, target1, feat2, target2):
        """
        双特征多目标优化

        输入: 两个特征名和目标值
        输出: 等值线交点（最优参数组合）
        """
        df = self.feature_df.copy()

        # 构建响应曲面
        radii = df['radius'].unique()
        sliders = df['slider'].unique()

        grid_r, grid_s = np.meshgrid(
            np.linspace(radii.min(), radii.max(), 50),
            np.linspace(sliders.min(), sliders.max(), 50)
        )

        # 插值曲面1
        surface1 = griddata(
            (df['radius'], df['slider']),
            df[feat1],
            (grid_r, grid_s),
            method='cubic'
        )

        # 插值曲面2
        surface2 = griddata(
            (df['radius'], df['slider']),
            df[feat2],
            (grid_r, grid_s),
